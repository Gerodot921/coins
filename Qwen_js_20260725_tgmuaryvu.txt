/* ---------- storage (in-memory Map + json file) ---------- */
let DB = {}; // { [userId]: UserState }
function loadDB() {
  try { DB = JSON.parse(fs.readFileSync(DATA_FILE, 'utf8')); }
  catch { DB = {}; }
  if (!DB || typeof DB !== 'object') DB = {};
  migrateEnergyCap();
}
function migrateEnergyCap() {
  // One-time migration: bump every existing user's energy cap to 100,000
  // and drop the old purchasable "energy" boost (energy is now fixed).
  const NEW_MAX = 100000;
  let changed = false;
  for (const id of Object.keys(DB)) {
    const u = DB[id];
    if (!u || typeof u !== 'object') continue;
    if ((u.maxEnergy || 0) < NEW_MAX) {
      const wasFull = u.energy >= (u.maxEnergy || 0);
      u.maxEnergy = NEW_MAX;
      u.energy = wasFull ? NEW_MAX : Math.min(NEW_MAX, u.energy || 0);
      u.regenPerSec = +(NEW_MAX / 86400).toFixed(4);
      if (u.boosts && u.boosts.energy) delete u.boosts.energy;
      changed = true;
    }
    if (u.referredBy === undefined) { u.referredBy = null; changed = true; }
  }
  if (changed) saveDB();
}
function saveDB() {
  const tmp = DATA_FILE + '.tmp';
  fs.writeFileSync(tmp, JSON.stringify(DB));
  fs.renameSync(tmp, DATA_FILE);
}
loadDB();

/* ---------- game constants & logic ---------- */
const RANKS = [
  { name: 'Bronze', from: 0 }, { name: 'Silver', from: 10000 },
  { name: 'Gold', from: 100000 }, { name: 'Platinum', from: 1000000 },
  { name: 'Diamond', from: 10000000 },
];
function rankFor(b) {
  let i = 0; for (let k = 0; k < RANKS.length; k++) if (b >= RANKS[k].from) i = k;
  const cur = RANKS[i], next = RANKS[i + 1];
  return { rank: cur.name, rankCurrent: b - cur.from,
    rankNext: next ? next.from - cur.from : cur.from,
    level: i + 1 + Math.floor(Math.log10(Math.max(1, b - cur.from) + 1)) };
}
function refreshRank(u) { const r = rankFor(u.balance); Object.assign(u, r); }
function recomputeEnergy(u, now = Date.now()) {
  const s = Math.max(0, (now - u.lastEnergyTs) / 1000);
  const g = Math.floor(s * u.regenPerSec);
  if (g > 0) { u.energy = Math.min(u.maxEnergy, u.energy + g); u.lastEnergyTs = now; }
}
function newUser(id, username) {
  const r = rankFor(0);
  const MAX_ENERGY = 100000;
  return { id, username, balance: 0, energy: MAX_ENERGY, maxEnergy: MAX_ENERGY, tapPower: 1,
    regenPerSec: +(MAX_ENERGY / 86400).toFixed(4), offlinePerHour: 0, extraRewardPct: 0,
    level: r.level, rank: r.rank, rankCurrent: r.rankCurrent, rankNext: r.rankNext,
    boosts: {}, tasksClaimed: [], airdropDone: [], friendsCount: 0, friendsEarned: 0,
    referredBy: null,
    lastEnergyTs: Date.now(), lastLoginTs: 0, createdAt: Date.now() };
}
function getUser(id, username) {
  let u = DB[id];
  if (!u) { u = newUser(id, username); DB[id] = u; saveDB(); }
  recomputeEnergy(u);
  return u;
}
function saveUser(u) { refreshRank(u); DB[u.id] = u; saveDB(); }

/* Real referral payouts: 15% / 5% / 3% of every tap's earnings flow up to
   the direct referrer, their referrer, and theirs. */
const REF_PCTS = [0.15, 0.05, 0.03];
function payReferralChain(u, amount) {
  if (amount <= 0) return;
  let current = u;
  for (let lvl = 0; lvl < REF_PCTS.length; lvl++) {
    if (!current.referredBy) break;
    const refUser = DB[current.referredBy];
    if (!refUser) break;
    const cut = Math.floor(amount * REF_PCTS[lvl]);
    if (cut > 0) {
      refUser.balance += cut;
      refUser.friendsEarned = (refUser.friendsEarned || 0) + cut;
      refreshRank(refUser);
    }
    current = refUser;
  }
}

/* ---------- tokens (HMAC-signed, no cookies → works cross-origin) ---------- */
const hmac = x => crypto.createHmac('sha256', SECRET).update(x).digest('hex');
const b64 = o => Buffer.from(JSON.stringify(o)).toString('base64url');
function signToken(id, username) {
  const p = b64({ id, username, exp: Date.now() + 1000 * 60 * 60 * 24 * 30 });
  return p + '.' + hmac(p);
}
function verifyToken(tok) {
  if (!tok) return null;
  const [p, sig] = tok.split('.'); if (!p || !sig) return null;
  try {
    if (!crypto.timingSafeEqual(Buffer.from(sig, 'hex'), Buffer.from(hmac(p), 'hex'))) return null;
  } catch { return null; }
  try {
    const d = JSON.parse(Buffer.from(p, 'base64url').toString());
    return d.exp > Date.now() ? { id: d.id, username: d.username } : null;
  } catch { return null; }
}

/* ---------- Telegram initData validation ---------- */
function validateInitData(initData) {
  if (!initData || !BOT_TOKEN) return null;
  const params = new URLSearchParams(initData);
  const hash = params.get('hash'); if (!hash) return null;
  params.delete('hash');
  const check = [...params.entries()].sort((a, b) => a[0].localeCompare(b[0])).map(([k, v]) => `${k}=${v}`).join('\n');
  const key = crypto.createHmac('sha256', 'WebAppData').update(BOT_TOKEN).digest();
  const calc = crypto.createHmac('sha256', key).update(check).digest('hex');
  if (calc !== hash) return null;
  const raw = params.get('user'); if (!raw) return null;
  try { const u = JSON.parse(raw); return { id: String(u.id), username: u.username }; }
  catch { return null; }
}

/* ---------- avatars (proxied through us so BOT_TOKEN never reaches the client) ---------- */
const AVATAR_TTL_OK = 6 * 60 * 60 * 1000;   // 6h cache for a found photo
const AVATAR_TTL_MISS = 15 * 60 * 1000;     // 15min cache for "no photo" (retry sooner)
const avatarCache = new Map();              // id -> { url: string|null, ts: number }

function tgApiGet(method, qs) {
  return new Promise((resolve, reject) => {
    const url = `https://api.telegram.org/bot${BOT_TOKEN}/${method}?${qs}`;
    https.get(url, r => {
      let data = '';
      r.on('data', c => { data += c; });
      r.on('end', () => { try { resolve(JSON.parse(data)); } catch (e) { reject(e); } });
    }).on('error', reject);
  });
}
function tgApiPost(method, payload) {
  return new Promise((resolve, reject) => {
    const body = JSON.stringify(payload);
    const req = https.request(`https://api.telegram.org/bot${BOT_TOKEN}/${method}`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json', 'Content-Length': Buffer.byteLength(body) },
    }, r => {
      let data = ''; r.on('data', c => { data += c; });
      r.on('end', () => { try { resolve(JSON.parse(data)); } catch (e) { reject(e); } });
    });
    req.on('error', reject); req.write(body); req.end();
  });
}

/* ---------- referral bot: long-poll for /start <refId> and reply with a
   Mini App button whose URL carries ?ref=<refId>, so the frontend can read
   it straight from location.search and register the referral. No webhook
   or extra hosting needed — this runs inside the same Node process. ---------- */
const MINIAPP_URL = process.env.MINIAPP_URL || ''; // e.g. https://your-app.vercel.app
let botUpdateOffset = 0;
async function pollBotUpdates() {
  if (!BOT_TOKEN) return; // no bot configured, nothing to poll
  try {
    const res = await tgApiGet('getUpdates', `offset=${botUpdateOffset}&timeout=25`);
    if (res && res.ok && Array.isArray(res.result)) {
      for (const upd of res.result) {
        botUpdateOffset = upd.update_id + 1;
        const msg = upd.message;
        if (!msg || !msg.text || !msg.text.startsWith('/start')) continue;
        const parts = msg.text.trim().split(/\s+/);
        const ref = parts[1] || '';
        const chatId = msg.chat.id;
        if (!MINIAPP_URL) {
          await tgApiPost('sendMessage', { chat_id: chatId, text: 'GIFT COIN бот работает, но MINIAPP_URL не настроен на сервере — обратитесь к администратору.' });
          continue;
        }
        const appUrl = MINIAPP_URL + (ref ? `?ref=${encodeURIComponent(ref)}` : '');
        await tgApiPost('sendMessage', {
          chat_id: chatId,
          text: 'Добро пожаловать в GIFT COIN! Жми кнопку ниже, чтобы начать 👇',
          reply_markup: { inline_keyboard: [[{ text: '🎮 Открыть GIFT COIN', web_app: { url: appUrl } }]] },
        });
      }
    }
  } catch (e) { /* network hiccup — just retry on the next tick */ }
  setTimeout(pollBotUpdates, 1000);
}
if (BOT_TOKEN) {
  tgApiGet('deleteWebhook', 'drop_pending_updates=false')
    .catch(() => {})
    .then(() => pollBotUpdates());
}

async function resolveAvatarFileUrl(id) {
  const cached = avatarCache.get(id);
  const ttl = cached && cached.url ? AVATAR_TTL_OK : AVATAR_TTL_MISS;
  if (cached && Date.now() - cached.ts < ttl) return cached.url;
  if (!BOT_TOKEN) return null;
  try {
    const photos = await tgApiGet('getUserProfilePhotos', `user_id=${encodeURIComponent(id)}&limit=1`);
    const sizes = photos && photos.ok && photos.result && photos.result.total_count > 0
      ? photos.result.photos[0] : null;
    const fileId = sizes && sizes[sizes.length - 1].file_id;
    if (!fileId) { avatarCache.set(id, { url: null, ts: Date.now() }); return null; }
    const fileInfo = await tgApiGet('getFile', `file_id=${encodeURIComponent(fileId)}`);
    const filePath = fileInfo && fileInfo.ok && fileInfo.result && fileInfo.result.file_path;
    if (!filePath) { avatarCache.set(id, { url: null, ts: Date.now() }); return null; }
    const url = `https://api.telegram.org/file/bot${BOT_TOKEN}/${filePath}`;
    avatarCache.set(id, { url, ts: Date.now() });
    return url;
  } catch {
    return null; // don't cache transient failures
  }
}

function serveAvatar(res, id) {
  resolveAvatarFileUrl(id).then(fileUrl => {
    if (!fileUrl) return send(res, 404, { error: 'no avatar' });
    https.get(fileUrl, tgRes => {
      if (tgRes.statusCode !== 200) { send(res, 404, { error: 'no avatar' }); return; }
      res.writeHead(200, {
        'Content-Type': tgRes.headers['content-type'] || 'image/jpeg',
        'Cache-Control': 'public, max-age=21600',
        'Access-Control-Allow-Origin': CORS_ORIGIN,
      });
      tgRes.pipe(res);
    }).on('error', () => send(res, 404, { error: 'no avatar' }));
  }).catch(() => send(res, 404, { error: 'no avatar' }));
}

/* ---------- http helpers ---------- */
function send(res, status, obj) {
  const body = JSON.stringify(obj);
  res.writeHead(status, {
    'Content-Type': 'application/json',
    'Access-Control-Allow-Origin': CORS_ORIGIN,
    'Access-Control-Allow-Methods': 'GET,POST,OPTIONS',
    'Access-Control-Allow-Headers': 'Authorization,Content-Type',
    'Cache-Control': 'no-store',
  });
  res.end(body);
}
function readBody(req) {
  return new Promise((resolve) => {
    let s = ''; req.on('data', c => { if (s.length < 1e6) s += c; });
    req.on('end', () => { try { resolve(s ? JSON.parse(s) : {}); } catch { resolve({}); } });
  });
}
function authed(req) {
  const h = req.headers['authorization'] || '';
  return h.startsWith('Bearer ') ? verifyToken(h.slice(7)) : null;
}

/* ---------- routes ---------- */
const STATIC = {
  '/': 'gift-coin-front.html',
  '/gift-coin-front.html': 'gift-coin-front.html',
  '/index.html': 'gift-coin-front.html',
};
function sendFile(res, filePath) {
  try {
    const body = fs.readFileSync(filePath);
    res.writeHead(200, {
      'Content-Type': 'text/html; charset=utf-8',
      'Cache-Control': 'no-store',
    });
    res.end(body);
  } catch {
    send(res, 404, { error: 'not found' });
  }
}

/* case / starter-case artwork, served straight from disk so the frontend
   can just point <img src> at ${API_BASE}/cases/<file>.png */
const CASES_DIR = path.join(__dirname, 'public', 'cases');
const IMG_TYPES = { '.png': 'image/png', '.jpg': 'image/jpeg', '.jpeg': 'image/jpeg', '.webp': 'image/webp' };
function serveCaseImage(res, name) {
  const safe = path.basename(String(name || ''));
  const filePath = path.join(CASES_DIR, safe);
  if (!filePath.startsWith(CASES_DIR)) return send(res, 400, { error: 'bad path' });
  fs.readFile(filePath, (err, body) => {
    if (err) return send(res, 404, { error: 'not found' });
    res.writeHead(200, {
      'Content-Type': IMG_TYPES[path.extname(filePath).toLowerCase()] || 'application/octet-stream',
      'Cache-Control': 'public, max-age=86400',
      'Access-Control-Allow-Origin': CORS_ORIGIN,
    });
    res.end(body);
  });
}

const server = http.createServer(async (req, res) => {
  if (req.method === 'OPTIONS') return send(res, 204, {});
  const url = req.url.split('?')[0];
  if (req.method === 'GET' && STATIC[url]) {
    return sendFile(res, path.join(__dirname, STATIC[url]));
  }
  if (req.method === 'GET' && url.startsWith('/avatar/')) {
    const id = decodeURIComponent(url.slice('/avatar/'.length));
    if (!id) return send(res, 404, { error: 'no avatar' });
    return serveAvatar(res, id);
  }
  if (req.method === 'GET' && url.startsWith('/cases/')) {
    return serveCaseImage(res, decodeURIComponent(url.slice('/cases/'.length)));
  }
  try {
    if (url === '/auth' && req.method === 'POST') {
      const { initData } = await readBody(req);
      const tg = validateInitData(initData || '');
      let id, username;
      if (tg) { id = tg.id; username = tg.username; }
      else if (AUTH_DEV) { id = 'dev-user'; username = 'dev'; }
      else return send(res, 401, { error: 'invalid initData' });
      const user = getUser(id, username); saveUser(user);
      return send(res, 200, { token: signToken(id, username), user });
    }

    const sess = authed(req);
    if (!sess) return send(res, 401, { error: 'unauthorized' });
    const u = getUser(sess.id, sess.username);

    if (url === '/me' && req.method === 'GET') { saveUser(u); return send(res, 200, { user: u }); }

    if (url === '/tap' && req.method === 'POST') {
      const { count } = await readBody(req);
      const n = Math.min(200, Math.max(0, Number(count) || 0));
      const taps = Math.min(n, u.energy);
      const gained = taps * u.tapPower;
      u.energy -= taps; u.balance += gained; u.lastEnergyTs = Date.now();
      payReferralChain(u, gained);
      saveUser(u); return send(res, 200, { user: u, gained });
    }

    if (url === '/ref/claim' && req.method === 'POST') {
      const { ref } = await readBody(req);
      const refId = ref !== undefined && ref !== null ? String(ref) : '';
      if (!refId || refId === String(u.id)) return send(res, 200, { ok: false, error: 'invalid ref', user: u });
      if (u.referredBy) return send(res, 200, { ok: false, error: 'already referred', user: u });
      const refUser = DB[refId];
      if (!refUser) return send(res, 200, { ok: false, error: 'referrer not found', user: u });
      u.referredBy = refId;
      refUser.friendsCount = (refUser.friendsCount || 0) + 1;
      const signupBonus = 100000;
      refUser.balance += signupBonus;
      refUser.friendsEarned = (refUser.friendsEarned || 0) + signupBonus;
      saveUser(u); saveUser(refUser);
      return send(res, 200, { ok: true, bonus: signupBonus, user: u });
    }

    if (url === '/claim' && req.method === 'POST') {
      const { taskId } = await readBody(req);
      const def = TASKS.find(t => t.id === taskId);
      if (!def) return send(res, 200, { ok: false, error: 'unknown task', user: u });
      const now = Date.now();
      if (def.kind === 'daily') {
        if (new Date(u.lastLoginTs).toDateString() === new Date(now).toDateString())
          return send(res, 200, { ok: false, error: 'already today', user: u });
        u.lastLoginTs = now;
      } else {
        if (u.tasksClaimed.includes(taskId)) return send(res, 200, { ok: false, error: 'already claimed', user: u });
        u.tasksClaimed.push(taskId);
      }
      const reward = Math.floor(def.reward * (1 + u.extraRewardPct));
      u.balance += reward;
      saveUser(u); return send(res, 200, { ok: true, reward, user: u });
    }

    if (url === '/boost' && req.method === 'POST') {
      const { boostId } = await readBody(req);
      const def = BOOSTS.find(b => b.id === boostId);
      if (!def) return send(res, 200, { ok: false, error: 'unknown boost', user: u });
      const lvl = u.boosts[boostId] || 0, cost = boostCost(def, lvl);
      if (u.balance < cost) return send(res, 200, { ok: false, error: 'not enough coins', user: u });
      u.balance -= cost; u.boosts[boostId] = lvl + 1;
      if (boostId === 'multitap') u.tapPower += 3;
      else if (boostId === 'regen') u.regenPerSec = +(u.regenPerSec * 1.25).toFixed(4);
      else if (boostId === 'offline') u.offlinePerHour += 100;
      else if (boostId === 'extra') u.extraRewardPct += 0.10;
      saveUser(u); return send(res, 200, { ok: true, user: u });
    }

    if (url === '/airdrop') {
      if (req.method === 'GET') return send(res, 200, { user: u, allocation: allocOf(u) });
      const { taskId } = await readBody(req);
      if (taskId && !u.airdropDone.includes(taskId)) u.airdropDone.push(taskId);
      saveUser(u); return send(res, 200, { user: u, allocation: allocOf(u) });
    }

    if (url === '/friends' && req.method === 'GET') {
      const list = Object.values(DB)
        .filter(x => x.referredBy === u.id)
        .sort((a, b) => (b.createdAt || 0) - (a.createdAt || 0))
        .map(x => ({ name: '@' + (x.username || ('user' + x.id)), reward: 100000, balance: x.balance || 0 }));
      return send(res, 200, { user: u, list });
    }

    if (url === '/leaderboard' && req.method === 'GET') {
      const rows = [
        { name: '@whale_01', score: 2500000000 }, { name: '@miner_x', score: 1800000000 },
        { name: '@tap_king', score: 1200000000 }, { name: '@you', score: u.balance, me: true },
        { name: '@player5', score: 850000000 }, { name: '@player6', score: 650000000 },
      ].sort((a, b) => b.score - a.score).map((r, i) => ({ rank: i + 1, name: r.name, score: fmt(r.score), me: r.me }));
      return send(res, 200, { rows });
    }

    if (url === '/shop') {
      if (req.method === 'GET') return send(res, 200, { user: u, packs: SHOP });
      return send(res, 200, { ok: false, error: 'payments coming soon', user: u }); // wire Stars/TON here
    }

    return send(res, 404, { error: 'not found' });
  } catch (e) {
    console.error(e); return send(res, 500, { error: 'server error' });
  }
});
